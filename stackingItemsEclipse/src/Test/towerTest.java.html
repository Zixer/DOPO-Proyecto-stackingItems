    @Test
    public void shouldSwapTwoObjectsInInsertionOrder() {
        Tower t = new Tower(30, 30);

        t.pushCup(5);
        t.pushCup(3);

        int pos5Before = -1;
        int pos3Before = -1;

        for (int i = 0; i < t.getInsertionOrder().size(); i++) {
            String[] item = t.getInsertionOrder().get(i);
            if (item[0].equals("cup") && item[1].equals("5")) {
                pos5Before = i;
            }
            if (item[0].equals("cup") && item[1].equals("3")) {
                pos3Before = i;
            }
        }

        t.swap(new String[]{"cup", "5"}, new String[]{"cup", "3"});

        int pos5After = -1;
        int pos3After = -1;

        for (int i = 0; i < t.getInsertionOrder().size(); i++) {
            String[] item = t.getInsertionOrder().get(i);
            if (item[0].equals("cup") && item[1].equals("5")) {
                pos5After = i;
            }
            if (item[0].equals("cup") && item[1].equals("3")) {
                pos3After = i;
            }
        }

        assertTrue(t.isOk());
        assertEquals(pos5Before, pos3After);
        assertEquals(pos3Before, pos5After);

        t.makeInvisible();
    }
